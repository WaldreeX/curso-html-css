        let categoriaActual = 'Todos';

        /**
         * Convierte segundos a formato MM:SS (ej. 12:45).
         */
        function formatearTiempo(segundos) {
            if (!segundos || isNaN(segundos) || !isFinite(segundos)) return '--:--';
            const min = Math.floor(segundos / 60);
            const seg = Math.floor(segundos % 60);
            return `${String(min).padStart(2, '0')}:${String(seg).padStart(2, '0')}`;
        }

        /**
         * Lee automáticamente los metadatos de cada archivo .mp4 para detectar su duración real.
         */
        function detectarDuraciones() {
            videosData.forEach(video => {
                const videoTemp = document.createElement('video');
                videoTemp.preload = 'metadata';
                videoTemp.src = video.url;

                videoTemp.onloadedmetadata = () => {
                    video.duracion = formatearTiempo(videoTemp.duration);
                    const badge = document.getElementById(`duracion-video-${video.id}`);
                    if (badge) {
                        badge.textContent = video.duracion;
                    }
                };
            });
        }

        /**
         * Cambia la categoría activa y actualiza el estilo de los botones de filtro.
         */
        function filtrarCategoria(categoria) {
            categoriaActual = categoria;
            const botones = document.querySelectorAll('.btn-filtro');

            botones.forEach(btn => {
                const cat = btn.getAttribute('data-cat');
                if (cat === categoria) {
                    if (cat === 'HTML5') {
                        btn.className = 'btn-filtro px-4 py-2 rounded-xl text-sm font-semibold bg-orange-500 text-white shadow-lg shadow-orange-500/20 transition-all';
                    } else if (cat === 'CSS3') {
                        btn.className = 'btn-filtro px-4 py-2 rounded-xl text-sm font-semibold bg-sky-500 text-white shadow-lg shadow-sky-500/20 transition-all';
                    } else {
                        btn.className = 'btn-filtro px-4 py-2 rounded-xl text-sm font-semibold bg-indigo-600 text-white shadow-lg shadow-indigo-500/20 transition-all';
                    }
                } else {
                    btn.className = 'btn-filtro px-4 py-2 rounded-xl text-sm font-semibold bg-slate-900 text-slate-300 border border-slate-800 hover:border-slate-700 hover:text-white transition-all';
                }
            });

            renderizarVideos();
        }

        /**
         * Abre el modal reproductor con estilo acorde al tema del video (HTML o CSS) y carga el .mp4 real.
         */
        function abrirReproductor(id) {
            const video = videosData.find(v => v.id === id);
            if (!video) return;

            const modal = document.getElementById('modal-reproductor');
            const caja = document.getElementById('modal-caja');
            const badge = document.getElementById('modal-badge');
            const glow = document.getElementById('modal-glow');
            const player = document.getElementById('reproductor-real');
            const errorBox = document.getElementById('error-video');
            const duracionModal = document.getElementById('modal-duracion');

            document.getElementById('modal-titulo').textContent = video.titulo;
            document.getElementById('modal-leccion').textContent = video.leccion;
            document.getElementById('modal-nivel').textContent = `Nivel ${video.nivel}`;
            document.getElementById('modal-descripcion').textContent = video.descripcion;

            badge.textContent = video.categoria;
            if (video.color === 'orange') {
                caja.className = 'bg-slate-900 border border-orange-500/40 rounded-3xl w-full max-w-4xl overflow-hidden shadow-2xl shadow-orange-500/10 transform transition-all';
                badge.className = 'px-2.5 py-1 rounded-full text-xs font-semibold bg-orange-500/15 text-orange-400 border border-orange-500/30 shrink-0';
                glow.className = 'absolute inset-0 bg-gradient-to-tr from-orange-500/20 via-transparent to-transparent pointer-events-none';
            } else {
                caja.className = 'bg-slate-900 border border-sky-500/40 rounded-3xl w-full max-w-4xl overflow-hidden shadow-2xl shadow-sky-500/10 transform transition-all';
                badge.className = 'px-2.5 py-1 rounded-full text-xs font-semibold bg-sky-500/15 text-sky-400 border border-sky-500/30 shrink-0';
                glow.className = 'absolute inset-0 bg-gradient-to-tr from-sky-500/20 via-transparent to-transparent pointer-events-none';
            }

            // Actualizar duración en el modal
            duracionModal.innerHTML = `
                <svg class="w-3.5 h-3.5 ${video.color === 'orange' ? 'text-orange-400' : 'text-sky-400'}" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                    <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M12 8v4l3 3m6-3a9 9 0 11-18 0 9 9 0 0118 0z"></path>
                </svg>
                ${video.duracion}
            `;

            // Cargar archivo .mp4 real
            errorBox.classList.add('hidden');
            player.classList.remove('hidden');
            player.src = video.url;

            player.onloadedmetadata = () => {
                const durReal = formatearTiempo(player.duration);
                video.duracion = durReal;
                duracionModal.innerHTML = `
                    <svg class="w-3.5 h-3.5 ${video.color === 'orange' ? 'text-orange-400' : 'text-sky-400'}" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                        <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M12 8v4l3 3m6-3a9 9 0 11-18 0 9 9 0 0118 0z"></path>
                    </svg>
                    ${durReal}
                `;
            };

            player.onerror = () => {
                player.classList.add('hidden');
                errorBox.classList.remove('hidden');
                document.getElementById('ruta-error').textContent = `Ruta configurada: "${video.url}"`;
            };

            modal.classList.remove('hidden');
            modal.classList.add('flex');
            player.play().catch(() => {});
        }

        /**
         * Cierra el modal reproductor y detiene la reproducción del video.
         */
        function cerrarReproductor() {
            const modal = document.getElementById('modal-reproductor');
            const player = document.getElementById('reproductor-real');
            player.pause();
            player.removeAttribute('src');
            player.load();
            modal.classList.add('hidden');
            modal.classList.remove('flex');
        }

        /**
         * Abre el menú lateral ocupando el 78% derecho y difumina el fondo.
         */
        function abrirMenuLateral() {
            const menuLateral = document.getElementById('menu-lateral');
            const backdrop = document.getElementById('menu-backdrop');
            
            menuLateral.classList.remove('translate-x-full');
            backdrop.classList.remove('opacity-0', 'pointer-events-none');
            backdrop.classList.add('opacity-100', 'pointer-events-auto');
        }

        /**
         * Cierra el menú lateral y quita el difuminado del fondo.
         */
        function cerrarMenuLateral() {
            const menuLateral = document.getElementById('menu-lateral');
            const backdrop = document.getElementById('menu-backdrop');
            
            menuLateral.classList.add('translate-x-full');
            backdrop.classList.remove('opacity-100', 'pointer-events-auto');
            backdrop.classList.add('opacity-0', 'pointer-events-none');
        }

        // Cerrar modal o menú con la tecla Escape
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                cerrarReproductor();
                cerrarMenuLateral();
            }
        });
